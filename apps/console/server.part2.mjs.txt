
    if (req.method === 'GET' && url.pathname === '/api/health') {
      const rpcOk = await rpcReachable();
      const llm = llmMeta();
      const e = env();
      return send(res, 200, {
        ok: rpcOk && Boolean(e.contractAddress),
        rpcOk,
        rpcUrl: e.rpcUrl,
        contractAddress: e.contractAddress || null,
        contractConfigured: Boolean(e.contractAddress),
        feeModel: `amount + ${feePercentLabel()} (budget checks totalCost)`,
        feeFormula: '(amount * 2) / 100',
        llm,
        demoAmounts: await getDemoAmounts(new ethers.JsonRpcProvider(RPC_URL)).catch(() => null),
        demoMerchants: resolveDemoMerchants(),
        settlement: {
          asset: process.env.PAYMENT_TOKEN || 'ETH',
          token: configuredErc20Address(),
          usdcSepolia: CIRCLE_USDC_ARB_SEPOLIA,
          note: configuredErc20Address()
            ? 'New grants escrow ERC-20 via grantSessionToken; proposeOrPay transfers the token to the merchant'
            : 'Native ETH escrow. Set PAYMENT_TOKEN=USDC (or USDG_TOKEN_ADDRESS) for stablecoin settlement.',
        },
        vercelHint:
          'Set RPC_URL, OWNER_PRIVATE_KEY, AGENT_PRIVATE_KEY, CONTRACT_ADDRESS, CONTRACT_DEPLOY_BLOCK, CHAIN_LABEL=arbitrum-sepolia on Vercel',
        arbitrumPath: {
          target: 'Arbitrum Sepolia (chainId 421614) — Open House Buildathon',
          checkScript: 'scripts/check-arbitrum-path.sh',
          deployScript: 'scripts/deploy-arbitrum-sepolia.sh',
          docs: 'docs/ARBITRUM_BUILDATHON.md',
          aaRoadmap: 'docs/AA_SESSION_KEY.md',
          usdgRoadmap: 'docs/USDG_ROADMAP.md',
          interface: 'contracts/src/interfaces/ISessionCapability.sol',
        },
        // Legacy field for older booth UI; prefer arbitrumPath
        furiosaPath: {
          note: 'legacy GWDC — use arbitrumPath for Arbitrum Buildathon',
          optionalLlm: 'LLM_PROVIDER=kiln|kimi (optional)',
          checkScript: 'scripts/check-furiosa-path.sh',
        },
        time: new Date().toISOString(),
      });
    }

    if (req.method === 'GET' && url.pathname === '/api/sessions') {
      const provider = new ethers.JsonRpcProvider(RPC_URL);
      const c = getContract(provider);
      const next = Number(await c.nextSessionId());
      const ids = [];
      for (let i = next - 1; i >= 0 && ids.length < 30; i--) ids.push(i);
      const merged = [...new Set([...grantedSessions, ...ids])].sort((a, b) => b - a);
      return send(res, 200, { nextSessionId: next, sessions: merged, grantedViaConsole: grantedSessions });
    }

    if (req.method === 'GET' && url.pathname === '/api/policy') {
      const sessionId = Number(url.searchParams.get('sessionId') || 0);
      const provider = new ethers.JsonRpcProvider(RPC_URL);
      const c = getContract(provider);
      const p = await c.getSessionPolicy(sessionId);
      // owner, agent, budget, spent, deadline, merchants, frozen, active
      const budget = p[2];
      const spent = p[3];
      const remaining = budget - spent;
      let token = ethers.ZeroAddress;
      try {
        token = await c.getSessionToken(sessionId);
      } catch (_) {}
      const meta = await readTokenMeta(provider, token);
      const fmt = (v) => formatUnitsAmount(v, meta.decimals);
      const llm = llmMeta();
      const now = Math.floor(Date.now() / 1000);
      const deadline = Number(p[4]);
      return send(res, 200, {
        rpcUrl: RPC_URL,
        contractAddress: CONTRACT_ADDRESS,
        feeModel: `Budget check uses amount + ${feePercentLabel()} fee`,
        feeFormula: '(amount * 2) / 100',
        ...llm,
        policy: {
          owner: p[0],
          agent: p[1],
          token: meta.token,
          symbol: meta.symbol,
          decimals: meta.decimals,
          budgetWei: budget.toString(),
          spentWei: spent.toString(),
          remainingWei: remaining.toString(),
          budgetEth: fmt(budget),
          spentEth: fmt(spent),
          remainingEth: fmt(remaining),
          spentPct: budget > 0n ? Number((spent * 10000n) / budget) / 100 : 0,
          remainingPct: budget > 0n ? Number((remaining * 10000n) / budget) / 100 : 0,
          deadline,
          deadlineIso: new Date(deadline * 1000).toISOString(),
          secondsLeft: Math.max(0, deadline - now),
          merchants: p[5],
          frozen: p[6],
          active: p[7],
        },
      });
    }

    if (req.method === 'POST' && url.pathname === '/api/grant') {
      const body = await readBody(req);
      const provider = new ethers.JsonRpcProvider(RPC_URL);
      const ownerWallet = new ethers.Wallet(OWNER_KEY, provider);
      const agentWallet = new ethers.Wallet(AGENT_KEY, provider);
      const agentAddress = body.agent || agentWallet.address;
      const c = getContract(ownerWallet);
      const amts = await getDemoAmounts(provider);
      const erc20Addr = configuredErc20Address();
      const duration = Number(body.durationSec || 3600);
      const merchants = body.merchants || [];
      let receipt;
      let settlement = { asset: 'ETH', token: null, decimals: 18 };
      if (erc20Addr) {
        const meta = await readTokenMeta(provider, erc20Addr);
        const budgetWei = parseUnitsAmount(String(body.budgetEth || amts.defaultGrant), meta.decimals);
        const erc20 = new ethers.Contract(erc20Addr, ERC20_APPROVE_ABI, ownerWallet);
        const bal = await erc20.balanceOf(ownerWallet.address);
        if (bal < budgetWei) {
          return send(res, 400, {
            error: `insufficient ${meta.symbol}: owner has ${formatUnitsAmount(bal, meta.decimals)} but grant needs ${formatUnitsAmount(budgetWei, meta.decimals)}. Faucet: https://faucet.circle.com/ (Arbitrum Sepolia USDC) or mint MockUSDG.`,
            balance: formatUnitsAmount(bal, meta.decimals),
            needed: formatUnitsAmount(budgetWei, meta.decimals),
            token: erc20Addr,
            faucet: 'https://faucet.circle.com/',
          });
        }
        const allowance = await erc20.allowance(ownerWallet.address, await c.getAddress());
        if (allowance < budgetWei) {
          const atx = await erc20.approve(await c.getAddress(), ethers.MaxUint256);
          await atx.wait();
        }
        const tx = await c.grantSessionToken(erc20Addr, budgetWei, duration, agentAddress, merchants);
        receipt = await tx.wait();
        settlement = { asset: meta.symbol, token: erc20Addr, decimals: meta.decimals };
      } else {
        const budgetWei = ethers.parseEther(String(body.budgetEth || amts.defaultGrant));
        {
          const bal = await provider.getBalance(ownerWallet.address);
          if (bal < budgetWei) {
            return send(res, 400, {
              error: `insufficient funds: owner has ${ethers.formatEther(bal)} ETH but grant needs ${ethers.formatEther(budgetWei)} ETH (plus gas). Top up Arbitrum Sepolia (or current RPC chain) or use a smaller budgetEth.`,
              balanceEth: ethers.formatEther(bal),
              neededEth: ethers.formatEther(budgetWei),
              suggestedBudgetEth: amts.autoGrant,
            });
          }
        }
        const tx = await c.grantSession(budgetWei, duration, agentAddress, merchants, { value: budgetWei });
        receipt = await tx.wait();
      }
      const next = await c.nextSessionId();
      const sessionId = Number(next) - 1;
      if (!grantedSessions.includes(sessionId)) grantedSessions.unshift(sessionId);
      const network = await provider.getNetwork();
      const chainId = Number(network.chainId);
      return send(res, 200, {
        transactionHash: receipt.hash,
        sessionId,
        agent: agentAddress,
        owner: ownerWallet.address,
        chain: resolveChainLabel(chainId),
        chainId,
        explorerUrl: explorerUrlForTx(chainId, receipt.hash),
        settlement,
        note: `Budget check will use amount + ${feePercentLabel()} fee`,
      });
    }

    if (req.method === 'POST' && url.pathname === '/api/freeze') {
      const body = await readBody(req);
      const provider = new ethers.JsonRpcProvider(RPC_URL);
      const wallet = new ethers.Wallet(OWNER_KEY, provider);
      const c = getContract(wallet);
      const tx = await c.freeze(Number(body.sessionId || 0));
      const receipt = await tx.wait();
      return send(res, 200, { transactionHash: receipt.hash, frozen: true });
    }


    if (req.method === 'POST' && url.pathname === '/api/close') {
      const body = await readBody(req);
      const provider = new ethers.JsonRpcProvider(RPC_URL);
      const wallet = new ethers.Wallet(OWNER_KEY, provider);
      const c = getContract(wallet);
      const sessionId = Number(body.sessionId || 0);
      const before = await c.getSessionPolicy(sessionId);
      // owner, agent, budget, spent, ... — refund on-chain = budget - spent + feesAccrued
      const remainingWei = before[2] - before[3];
      const tx = await c.closeSession(sessionId);
      const receipt = await tx.wait();
      return send(res, 200, {
        transactionHash: receipt.hash,
        closed: true,
        refundedEth: ethers.formatEther(remainingWei > 0n ? remainingWei : 0n),
        note: 'Session closed; remaining budget refunded to owner (AI Sovereignty)',
      });
    }

    if (req.method === 'POST' && url.pathname === '/api/propose') {
      const body = await readBody(req);
      const { paymentProposer } = await initializeAgent();
      const opts = { explain: body.explain !== false };
      if (body.forceAmountEth !== undefined) opts.forceAmountEth = String(body.forceAmountEth);
      if (body.forceMerchant !== undefined) opts.forceMerchant = String(body.forceMerchant);
      if (body.allowOffAllowlist) opts.allowOffAllowlist = true;
      if (body.skipLlm === true) opts.skipLlm = true;
      const result = await paymentProposer.proposePayment(
        Number(body.sessionId || 0),
        body.intent || (await getDemoAmounts(new ethers.JsonRpcProvider(RPC_URL))).autoProposeIntent,
        opts
      );
      return send(res, 200, result);
    }

    if (req.method === 'POST' && url.pathname === '/api/demo/boundary') {
      const body = await readBody(req);
      const which = (body.case || 'budget').toLowerCase();
      // Default: real LLM propose, then Guard deny (dual evidence).
      // Pass skipLlm:true for a fast forced/adversarial path (no LLM).
      const skipLlm = body.skipLlm === true;
      const { paymentProposer } = await initializeAgent();
      const provider = new ethers.JsonRpcProvider(RPC_URL);
      const wallet = new ethers.Wallet(OWNER_KEY, provider);
      const c = getContract(wallet);
      const amts = await getDemoAmounts(provider);
      const unit = amts.unit || 'ETH';

      let budgetEth;
      let merchants;
      let forceAmountEth;
      let forceMerchant;
      let allowOffAllowlist;
      let intent;

      const m = resolveDemoMerchants();
      if (which === 'merchant') {
        budgetEth = amts.merchantBudget;
        merchants = [m.coffee];
        forceAmountEth = amts.merchantForce || amts.coffeeQuote || '0.01';
        forceMerchant = m.shadow;
        allowOffAllowlist = true;
        intent = skipLlm
          ? 'Forced adversarial: pay off-allowlist merchant'
          : `Agentic Commerce boundary: please pay ${forceAmountEth} ${unit} to merchant ${m.shadow} (not on allowlist)`;
      } else {
        // forceAmount + 2% fee must exceed budget (deny = success)
        budgetEth = amts.boundaryBudget;
        merchants = [m.coffee];
        forceAmountEth = amts.boundaryForce;
        forceMerchant = m.coffee;
        allowOffAllowlist = false;
        intent = skipLlm
          ? 'Forced adversarial: over-budget (amount + 2% fee)'
          : `Boundary demo: buy premium item for ${forceAmountEth} ${unit} (session budget is tight — expect Guard deny if over budget)`;
      }

      const erc20Addr = configuredErc20Address();
      const decimals = erc20Addr ? (amts.decimals || 6) : 18;
      const budgetWei = parseUnitsAmount(budgetEth, decimals);
      const agentWallet = new ethers.Wallet(AGENT_KEY, provider);
      if (erc20Addr) {
        const erc20 = new ethers.Contract(erc20Addr, ERC20_APPROVE_ABI, wallet);
        const allowance = await erc20.allowance(wallet.address, await c.getAddress());
        if (allowance < budgetWei) {
          const atx = await erc20.approve(await c.getAddress(), ethers.MaxUint256);
          await atx.wait();
        }
        const tx = await c.grantSessionToken(erc20Addr, budgetWei, 3600, agentWallet.address, merchants);
        await tx.wait();
      } else {
        const tx = await c.grantSession(budgetWei, 3600, agentWallet.address, merchants, { value: budgetWei });
        await tx.wait();
      }
      const next = await c.nextSessionId();
      const sessionId = Number(next) - 1;
      if (!grantedSessions.includes(sessionId)) grantedSessions.unshift(sessionId);

      const amountWei = parseUnitsAmount(forceAmountEth, decimals);
      // LLM path: proposePayment still calls the LLM (usage in result), then applies
      // forceAmountEth/forceMerchant so the on-chain deny is deterministic for booth demos.
      const result = await paymentProposer.proposePayment(sessionId, intent, {
        forceAmountEth,
        forceMerchant,
        allowOffAllowlist,
        skipLlm,
        explain: true,
      });

      return send(res, 200, {
        case: which,
        mode: skipLlm ? 'forced-skip-llm' : 'llm-propose-then-guard',
        skipLlm,
        sessionId,
        settlement: { asset: unit, token: erc20Addr, decimals },
        feePreview: {
          amountEth: forceAmountEth,
          feeEth: formatUnitsAmount(feeWei(amountWei), decimals),
          totalCostEth: formatUnitsAmount(totalCostWei(amountWei), decimals),
          feePercent: feePercentLabel(),
          note: 'Deny is a success outcome — contract enforced the boundary',
        },
        dualEvidence: {
          llm: skipLlm
            ? 'skipped (forced adversarial)'
            : 'real LLM propose — see result.usage.propose + result.proposal.reasoning',
          chain: 'PaymentDenied on SessionPolicy (Guard success)',
        },
        result,
      });
    }

    if (req.method === 'GET' && url.pathname === '/api/events') {
      const sessionId = Number(url.searchParams.get('sessionId') || 0);
      const { policyReader } = await initializeAgent();
      const events = await policyReader.getPaymentHistory(sessionId, 0);
      let decimals = 18;
      try {
        const pol = await policyReader.getSessionPolicy(sessionId);
        decimals = pol.decimals ?? 18;
      } catch (_) {}
      return send(res, 200, {
        feeModel: `amount + ${feePercentLabel()}`,
        events: events.map((e) => enrichEvent(e, decimals)),
      });
    }

    send(res, 404, { error: 'not found' });
  } catch (err) {
    send(res, 500, { error: err.message || String(err) });
  }
}

export default handler;

    console.log(`Contract ${e.contractAddress || '(unset)'}`);
    console.log(`Fee model: amount + ${feePercentLabel()} (budget checks totalCost)`);
  });
